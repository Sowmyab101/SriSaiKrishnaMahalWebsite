// ENQUIRES
// Load Enquiries
function loadEnquiries() {
  const table = document.querySelector("#enquiryTable tbody");

  // ✅ STOP if table not found (important)
  if (!table) return;

  const enquiries = JSON.parse(localStorage.getItem("enquiries")) || [];

  table.innerHTML = "";

  enquiries.forEach((e) => {
    let row = `
      <tr>
        <td>${e.name}</td>
        <td>${e.phone}</td>
        <td>${e.message}</td>
      </tr>
    `;
    table.innerHTML += row;
  });
}

loadEnquiries();

function downloadEnquiryExcel() {
  const enquiries = JSON.parse(localStorage.getItem("enquiries")) || [];

  let csv = "Name,Phone,Message\n";

  enquiries.forEach(e => {
    csv += `${e.name},${e.phone},${e.message}\n`;
  });

  const blob = new Blob([csv], { type: "text/csv" });
  const link = document.createElement("a");

  link.href = URL.createObjectURL(blob);
  link.download = "enquiries.csv";
  link.click();
}
